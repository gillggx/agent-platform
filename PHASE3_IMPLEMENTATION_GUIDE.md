```python
# At step execution start
async def execute_step(self, step_id: str, step_def: Dict) -> StepOutput:
    await progress_tracker.on_step_started(
        workflow_id=self.workflow_id,
        workflow_run_id=self.workflow_run_id,
        step_id=step_id,
        role=step_def.get("role", "unknown")
    )
    
    try:
        # ... execute step ...
        output = result
        
        # At step completion
        await progress_tracker.on_step_completed(
            workflow_id=self.workflow_id,
            workflow_run_id=self.workflow_run_id,
            step_id=step_id,
            role=step_def.get("role", "unknown"),
            output=output
        )
        
        return output
        
    except Exception as e:
        await progress_tracker.on_workflow_failed(
            workflow_id=self.workflow_id,
            workflow_run_id=self.workflow_run_id,
            error=str(e)
        )
        raise
```

### Step 2: Save Agent Output as Artifact
```python
async def save_agent_output_as_artifact(
    workflow_id: str,
    workflow_run_id: str,
    step_id: str,
    agent_role: str,
    output_content: str,
    artifact_type: str = "product_spec"
) -> str:
    """Save agent output as an artifact and emit progress event."""
    
    # Create artifact metadata
    metadata = ArtifactMetadata(
        title=f"{artifact_type} - {agent_role}",
        author=agent_role,
        workflow_run_id=workflow_run_id,
        step_id=step_id,
        description=f"Generated by {agent_role} in step {step_id}"
    )
    
    # Create artifact
    artifact = ArtifactSchema(
        workflow_run_id=workflow_run_id,
        artifact_type=ArtifactType(artifact_type),
        content=output_content,
        metadata=metadata
    )
    
    # Save to store
    artifact_id = await artifact_store.save_artifact(workflow_run_id, artifact)
    
    # Emit progress event (content preview)
    preview = output_content[:500] if len(output_content) > 500 else output_content
    await progress_tracker.on_agent_output(
        workflow_id=workflow_id,
        workflow_run_id=workflow_run_id,
        step_id=step_id,
        role=agent_role,
        content=preview
    )
    
    return artifact_id
```

### Step 3: Support Document Download
```python
# When agent response is ready to export
```

## JavaScript/React WebSocket Client Example

```javascript
import React, { useEffect, useState } from 'react';

export function WorkflowProgressPanel({ workflowId }) {
  const [events, setEvents] = useState([]);
  const [status, setStatus] = useState('connecting');
  const [artifacts, setArtifacts] = useState([]);

  useEffect(() => {
    const ws = new WebSocket(
      `ws://localhost:8000/ws/workflows/${workflowId}`
    );

    ws.onopen = () => {
      setStatus('connected');
      console.log('Connected to workflow progress stream');
    };

    ws.onmessage = (event) => {
      try {
        const message = JSON.parse(event.data);
        
        // Skip connection confirmation message
        if (message.type === 'connection') {
          return;
        }

        // Add to event list
        setEvents(prev => [...prev, message]);

        // Handle specific event types
        switch(message.event_type) {
          case 'step_started':
            console.log(`⏳ ${message.role}: ${message.content}`);
            break;

          case 'step_completed':
            console.log(
              `✅ ${message.role}: Completed in ${message.metadata.duration_ms}ms`
            );
            break;

          case 'agent_output':
            console.log(`📝 ${message.role}:\n${message.content}`);
            break;

          case 'workflow_completed':
            setStatus('completed');
            setArtifacts(message.metadata.artifacts || []);
            console.log('✨ Workflow completed!');
            console.log('Artifacts:', message.metadata.artifacts);
            break;

          case 'workflow_failed':
            setStatus('failed');
            console.error('❌ Workflow failed:', message.content);
            break;

          default:
            console.log('Event:', message.event_type);
        }
      } catch (e) {
        console.error('Error parsing message:', e);
      }
    };

    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
      setStatus('error');
    };

    ws.onclose = () => {
      setStatus('disconnected');
      console.log('Disconnected from progress stream');
    };

    return () => {
      ws.close();
    };
  }, [workflowId]);

  return (
    <div className="progress-panel">
      <h2>Workflow Progress</h2>
      <p>Status: <span className={`status ${status}`}>{status}</span></p>

      <div className="events-log">
        {events.map((event, idx) => (
          <div key={idx} className="event">
            <span className="timestamp">{event.timestamp}</span>
            <span className="type">[{event.event_type}]</span>
            <span className="role">{event.role}</span>
            <p>{event.content}</p>
          </div>
        ))}
      </div>

      {artifacts.length > 0 && (
        <div className="artifacts">
          <h3>Generated Artifacts</h3>
          <ul>
            {artifacts.map(id => (
              <li key={id}>
                <a href={`/api/v1/artifacts/${id}/download`}>
                  Download {id}
                </a>
              </li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}
```

## Performance Considerations

### Document Generation
- Markdown parsing: ~1-5ms per KB
- DOCX generation: ~10-50ms per artifact
- Memory: ~1-2MB per artifact in memory

### WebSocket
- Connection overhead: ~1-2ms per client
- Broadcast latency: <10ms for 100 clients
- Memory per client: ~50-100KB

### Artifact Storage
- In-memory storage suitable for:
  - Single server deployments
  - Development/testing
  - <10,000 artifacts
- For production with high volume:
  - Migrate to database (PostgreSQL)
  - Add artifact archival
  - Implement cleanup policies

## Troubleshooting

### WebSocket Connection Fails
```
Error: Connection refused
```
**Solution:** Ensure server is running and WebSocket endpoint is accessible
```bash
curl -i http://localhost:8000/health
```

### DOCX Generation Fails
```
Error: Failed to generate document
```
**Solution:** Check artifact content is valid markdown
```python
# Test parser
from app.output.document_generator import MarkdownParser
parser = MarkdownParser()
```
